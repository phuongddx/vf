# vf Codegen Hybrid + Full Render Implementation Plan (B2)

> **For agentic workers:** REQUIRED SUB-SKILL: Use text2prod:subagent-driven-development (recommended) or text2prod:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `vf make` renders a REAL mp4 end-to-end — LLM spec → LLM custom-scene codegen (tsc-validated, template fallback) → `npm install` → `npx remotion render` → ffmpeg post — plus a real vf-generated demo video committed to the repo and embedded in the README.

**Architecture:** Scaffold becomes a working data-driven Remotion project (template Player always renders any spec; custom scenes overlay via a registry map). Codegen stage reuses B's provider machinery. npm/npx/tsc/ffmpeg all run through the `Executor` trait (mocked in tests). Render failures degrade: manual command printed, stage loop breaks, exit 0.

**Tech Stack:** Rust 1.85/edition 2024, Remotion 4.0.539 (`remotion` + `@remotion/cli`), React 19.3.1, TypeScript 7.0.2, wiremock, assert_cmd.

**Spec:** `docs/specs/2026-10-31-codegen-design.md` (same repo)

pub fn codegen_skip(lang: Lang) -> String {

}

/// Regenerate src/scenes/registry.ts from existing Custom_s*.tsx files.
/// Returns the number of custom components registered.
pub fn rebuild_registry(project_dir: &Path) -> anyhow::Result<usize> {
    let scenes_dir = project_dir.join("src").join("scenes");
    let mut ids: Vec<String> = Vec::new();
    if scenes_dir.exists() {
        for entry in std::fs::read_dir(&scenes_dir)? {
            let entry = entry?;
            let name = entry.file_name().to_string_lossy().into_owned();
            if let Some(id) = name
                .strip_prefix("Custom_")
                .and_then(|s| s.strip_suffix(".tsx"))
            {
                ids.push(id.to_string());
            }
        }
    }
    ids.sort();
    let mut out = String::from(
        "import type { FC } from \"react\";\n\nexport interface SceneLike {\n  id: string;\n  duration: number;\n  custom?: boolean;\n  narration?: string | null;\n  elements?: Array<{ type: string; content?: string; src?: string; motion?: string | null }>;\n}\n\nexport interface SceneProps {\n  scene: SceneLike;\n  palette: string[];\n  font: string;\n}\n\n// vf regenerates this file by scanning src/scenes/Custom_*.tsx — do not edit.\n",
    );
    for id in &ids {
        out.push_str(&format!("import Custom_{id} from \"./Custom_{id}\";\n"));
    }
    out.push_str("\nexport const customScenes: Record<string, FC<SceneProps>> = {\n");
    for id in &ids {
        out.push_str(&format!("  \"{id}\": Custom_{id},\n"));
    }
    out.push_str("};\n");
    std::fs::create_dir_all(&scenes_dir)?;
    std::fs::write(scenes_dir.join("registry.ts"), out)?;
    Ok(ids.len())
}

        .or_else(|| t.strip_prefix("```"))

    let t = t.trim();
    if t.contains("export default") {

async fn tsc_passes(exec: &dyn crate::exec::Executor, project_dir: &std::path::Path) -> Option<String> {
    if std::env::var("VF_TSC").map(|v| v == "0").unwrap_or(false) {
        return Some(String::new()); // explicitly disabled — treat as pass
    }
    let cmd = crate::exec::CommandSpec {
        program: "npx".into(),
        args: vec!["tsc".into(), "--noEmit".into()],
        cwd: Some(project_dir.to_path_buf()),
    };
    match exec.run(&cmd).await {
        Ok(out) if out.status == 0 => Some(String::new()),
        Ok(out) => None::<String>.or(Some(out.stderr)).filter(|_| true).map(|_| out.stderr.clone()),
        Err(_) => Some(String::new()), // no node/npx — light check only
    }
}

pub async fn codegen_custom_scenes(
    cfg: &crate::config::Config,
    spec: &crate::ir::VideoSpec,
    project_dir: &std::path::Path,
    exec: &dyn crate::exec::Executor,
) -> CodegenOutcome {
    use crate::llm::LlmClient as _;
    let custom: Vec<&crate::ir::Scene> = spec.scenes.iter().filter(|s| s.custom).collect();
    if custom.is_empty() {
        return CodegenOutcome::Skipped;
    }
    let Some(provider_name) = cfg.routing.codegen.clone() else {
        return CodegenOutcome::Skipped;
    };
    let Some(p) = cfg.providers.get(&provider_name) else {
        return CodegenOutcome::Skipped;
    };
    let key_missing = std::env::var(&p.api_key_env).map(|v| v.is_empty()).unwrap_or(true);
    if key_missing || crate::llm::check_tls(&p.base_url).is_err() {
        return CodegenOutcome::Skipped;
    }
    let Ok(inner) = crate::llm::registry::build(&provider_name, cfg) else {
        return CodegenOutcome::Skipped;
    };
    let client = crate::llm::retry::RetryingClient::new(inner);
    let mut usage_total: Option<crate::llm::Usage> = None;
    let mut written = 0usize;
    let mut fallbacks: Vec<String> = Vec::new();
    for scene in custom {
        let path = project_dir.join("src/scenes").join(format!("Custom_{}.tsx", scene.id));
        let mut feedback: Option<String> = None;
        let mut done = false;
        for _attempt in 0..2 {
            let req = codegen_request(scene, &spec.style, feedback.as_deref());
            match client.complete(&req).await {
                Err(_) => break,
                Ok(resp) => {
                    if let Some(u) = resp.usage {
                        usage_total = Some(match usage_total {
                            None => u,
                            Some(prev) => crate::llm::Usage {
                                input_tokens: prev.input_tokens + u.input_tokens,
                                output_tokens: prev.output_tokens + u.output_tokens,
                            },
                        });
                    }
                    if let Some(code) = extract_code(&resp.text) {
                        if let Some(parent) = path.parent() {
                            let _ = std::fs::create_dir_all(parent);
                        }
                        let _ = std::fs::write(&path, &code);
                        match tsc_passes(exec, project_dir).await {
                            Some(_) => { written += 1; done = true; break; }
                            None => feedback = Some("TypeScript compile failed".into()),
                        }
                    } else {
                        feedback = Some("no export default found".into());
                    }
                }
            }
        }
        if !done {
            let _ = std::fs::remove_file(&path);
            fallbacks.push(scene.id.clone());
        }
    }
    let _ = crate::engine::scaffold::rebuild_registry(project_dir);
    CodegenOutcome::Done { usage: usage_total, written, fallbacks }
}
